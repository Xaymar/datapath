6//! The public API: contexts, sockets and servers.
7//!
8//! Handlers are given to a builder before any I/O starts, so no message can
9//! arrive before its handler exists.
10//!
11//! ```no_run
12//! use datapath_core::{Server, Socket};
13//!
14//! let server = Server::host("my-service")
15//!     .on_accept(|incoming| {
16//!         let socket = incoming
17//!             .on_message(|msg| println!("got {} bytes", msg.len()))
18//!             .accept();
19//!         // Keep `socket` alive for as long as the connection should stay open.
20//!         std::mem::forget(socket);
21//!     })
22//!     .open()?;
23//!
24//! let client = Socket::connect("my-service").open()?;
25//! client.send(b"hello")?;
26//! # drop(server);
27//! # Ok::<(), Box<dyn std::error::Error>>(())
28//! ```
29

38/// Owns the I/O threads. Cloning gives another handle; when the last handle is
39/// dropped, every socket and server created on the context is closed (sockets
40/// and servers themselves do not keep their context alive).
41#[derive(Clone)]
42pub struct Context {
43    handle: Arc<sys::ContextHandle>,
44}
45
46impl Context {
47    /// Start building a context.
48    pub fn builder() -> ContextBuilder {
49        ContextBuilder::default()
50    }
51
52    /// A context with the default options.
53    pub fn new() -> Result<Context, Error> {
54        Context::builder().build()
55    }
56
57    /// The process-wide default context, used when none is given.
58    pub fn global() -> Result<Context, Error> {
59        static GLOBAL: OnceLock<Context> = OnceLock::new();
60        if let Some(c) = GLOBAL.get() {
61            return Ok(c.clone());
62        }
63        let c = Context::new()?;
64        Ok(GLOBAL.get_or_init(|| c).clone())
65    }
66
67    fn shared(&self) -> &Arc<sys::ContextShared> {
68        self.handle.shared()
69    }
70}

135/// One end of a connection. Dropping it closes the connection; no handler runs
136/// after the drop returns.
137pub struct Socket {
138    conn: Arc<sys::Conn>,
139}
140
141impl Socket {
142    /// Connect to the server at `path`.
143    pub fn connect(path: impl Into<String>) -> ConnectBuilder {
144        ConnectBuilder {
145            path: path.into(),
146            context: None,
147            opts: Options::default(),
148            on_message: None,
149            on_close: None,
150        }
151    }
152
153    /// Queue one message. Never blocks; safe to call from handlers.
154    pub fn send(&self, data: &[u8]) -> Result<(), SendError> {
155        self.conn.send(&[data])
156    }
157
158    /// Queue one message assembled from several parts.
159    pub fn send_vectored(&self, parts: &[IoSlice<'_>]) -> Result<(), SendError> {
160        send_vectored(&self.conn, parts)
161    }
162
163    /// A cheap, cloneable handle that can only send. It does not keep the
164    /// connection alive: once this socket is dropped, sending returns
165    /// [`SendError::Closed`].
166    pub fn sender(&self) -> Sender {
167        Sender {
168            conn: Arc::downgrade(&self.conn),
169        }
170    }
171
172    /// Replace the message handler. The first handler starts receiving; until
173    /// then the library does not read, so it also does not notice the peer closing.
174    pub fn set_on_message(&self, handler: impl FnMut(&[u8]) + Send + 'static) {
175        self.conn.set_message_handler(Some(Box::new(handler)));
176    }
177
178    /// Replace the close handler.
179    pub fn set_on_close(&self, handler: impl FnOnce(CloseReason) + Send + 'static) {
180        self.conn.set_close_handler(Some(Box::new(handler)));
181    }
182
183    /// Close the connection; the close handler fires. Returns false if it was
184    /// closed already.
185    pub fn close(&self) -> bool {
186        self.conn.close()
187    }
188
189    /// Remove the message handler. Reading, once started, continues.
190    pub fn clear_on_message(&self) {
191        self.conn.set_message_handler(None);
192    }
193
194    /// Remove the close handler.
195    pub fn clear_on_close(&self) {
196        self.conn.set_close_handler(None);
197    }
198
199    /// True while the connection is open.
200    pub fn is_open(&self) -> bool {
201        self.conn.is_open()
202    }
203
204    /// The negotiated message size limit of this connection.
205    pub fn max_message_size(&self) -> u64 {
206        self.conn.max_message_size()
207    }
208}
209
210impl Drop for Socket {
211    fn drop(&mut self) {
212        self.conn.release();
213    }
214}
215
216impl std::fmt::Debug for Socket {
217    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
218        f.debug_struct("Socket")
219            .field("open", &self.is_open())
220            .finish()
221    }
222}

224fn send_vectored(conn: &sys::Conn, parts: &[IoSlice<'_>]) -> Result<(), SendError> {
225    const INLINE: usize = 8;
226    if parts.len() <= INLINE {
227        let mut buf: [&[u8]; INLINE] = [&[]; INLINE];
228        for (slot, p) in buf.iter_mut().zip(parts) {
229            *slot = p;
230        }
231        conn.send(&buf[..parts.len()])
232    } else {
233        let all: Vec<&[u8]> = parts.iter().map(|p| &**p).collect();
234        conn.send(&all)
235    }
236}

238/// The sending half of a [`Socket`]; see [`Socket::sender`].
239#[derive(Clone)]
240pub struct Sender {
241    conn: Weak<sys::Conn>,
242}
243
244impl Sender {
245    /// Queue one message.
246    pub fn send(&self, data: &[u8]) -> Result<(), SendError> {
247        match self.conn.upgrade() {
248            Some(c) => c.send(&[data]),
249            None => Err(SendError::Closed),
250        }
251    }
252
253    /// Queue one message assembled from several parts.
254    pub fn send_vectored(&self, parts: &[IoSlice<'_>]) -> Result<(), SendError> {
255        match self.conn.upgrade() {
256            Some(c) => send_vectored(&c, parts),
257            None => Err(SendError::Closed),
258        }
259    }
260}

262/// Builds a client [`Socket`].
263pub struct ConnectBuilder {
264    path: String,
265    context: Option<Context>,
266    opts: Options,
267    on_message: Option<MessageHandler>,
268    on_close: Option<CloseHandler>,
269}
270
271impl ConnectBuilder {
272    /// Run on this context instead of the global one.
273    pub fn context(mut self, context: &Context) -> Self {
274        self.context = Some(context.clone());
275        self
276    }
277
278    /// Use these options as a whole.
279    pub fn options(mut self, opts: Options) -> Self {
280        self.opts = opts;
281        self
282    }
283
284    /// Largest message accepted.
285    pub fn max_message_size(mut self, bytes: u64) -> Self {
286        self.opts.max_message_size = bytes;
287        self
288    }
289
290    /// How long to wait for the handshake.
291    pub fn timeout(mut self, timeout: Duration) -> Self {
292        self.opts.connect_timeout = timeout;
293        self
294    }
295
296    /// Handler for received messages. Runs on an I/O thread, never concurrently
297    /// with itself or with the close handler of the same socket.
298    pub fn on_message(mut self, handler: impl FnMut(&[u8]) + Send + 'static) -> Self {
299        self.on_message = Some(Box::new(handler));
300        self
301    }
302
303    /// Handler for the end of the connection; runs at most once.
304    pub fn on_close(mut self, handler: impl FnOnce(CloseReason) + Send + 'static) -> Self {
305        self.on_close = Some(Box::new(handler));
306        self
307    }
308
309    /// Connect. Blocks until the handshake completes or the timeout passes.
310    pub fn open(self) -> Result<Socket, Error> {
311        let context = match self.context {
312            Some(c) => c,
313            None => Context::global()?,
314        };
315        let conn =
316            context
317                .shared()
318                .connect(&self.path, &self.opts, self.on_message, self.on_close)?;
319        Ok(Socket { conn })
320    }
321}

323/// A hosted server. Dropping it stops accepting and closes its connections.
324pub struct Server {
325    listener: Arc<sys::Listener>,
326}
327
328impl Server {
329    /// Host a server at `path`.
330    pub fn host(path: impl Into<String>) -> HostBuilder {
331        HostBuilder {
332            path: path.into(),
333            context: None,
334            opts: Options::default(),
335            on_accept: None,
336        }
337    }
338
339    /// Stop accepting and close every connection of this server. Returns false
340    /// if it was closed already.
341    pub fn close(&self) -> bool {
342        self.listener.close()
343    }
344
345    /// Replace the accept handler. Clients that passed the handshake before
346    /// any handler was set are offered to it now.
347    pub fn set_on_accept(&self, handler: impl FnMut(Incoming) + Send + 'static) {
348        set_accept(&self.listener, Box::new(handler));
349    }
350
351    /// Remove the accept handler: later clients wait until one is set.
352    pub fn clear_on_accept(&self) {
353        self.listener.set_accept_handler(None);
354    }
355}
356
357fn set_accept(listener: &sys::Listener, mut f: Box<dyn FnMut(Incoming) + Send + 'static>) {
358    listener.set_accept_handler(Some(Box::new(move |conn| f(Incoming { conn: Some(conn) }))));
359}
360
361impl Drop for Server {
362    fn drop(&mut self) {
363        self.listener.release();
364    }
365}

367/// Builds a [`Server`].
368pub struct HostBuilder {
369    path: String,
370    context: Option<Context>,
371    opts: Options,
372    on_accept: Option<Box<dyn FnMut(Incoming) + Send + 'static>>,
373}
374
375impl HostBuilder {
376    /// Run on this context instead of the global one.
377    pub fn context(mut self, context: &Context) -> Self {
378        self.context = Some(context.clone());
379        self
380    }
381
382    /// Use these options as a whole.
383    pub fn options(mut self, opts: Options) -> Self {
384        self.opts = opts;
385        self
386    }
387
388    /// Who may connect.
389    pub fn permissions(mut self, permissions: Permissions) -> Self {
390        self.opts.permissions = permissions;
391        self
392    }
393
394    /// Largest message accepted.
395    pub fn max_message_size(mut self, bytes: u64) -> Self {
396        self.opts.max_message_size = bytes;
397        self
398    }
399
400    /// Maximum connected clients; 0 = unlimited.
401    pub fn max_clients(mut self, n: usize) -> Self {
402        self.opts.max_clients = n;
403        self
404    }
405
406    /// Handler for clients that passed the handshake. Runs on an I/O thread.
407    /// Accept an [`Incoming`] to get a [`Socket`]; drop it to reject. Clients
408    /// that arrive before a handler is set wait.
409    pub fn on_accept(mut self, handler: impl FnMut(Incoming) + Send + 'static) -> Self {
410        self.on_accept = Some(Box::new(handler));
411        self
412    }
413
414    /// Start hosting.
415    pub fn open(self) -> Result<Server, Error> {
416        let context = match self.context {
417            Some(c) => c,
418            None => Context::global()?,
419        };
420        let listener = context.shared().listen(&self.path, &self.opts)?;
421        if let Some(mut f) = self.on_accept {
422            listener
423                .set_accept_handler(Some(Box::new(move |conn| f(Incoming { conn: Some(conn) }))));
424        }
425        Ok(Server { listener })
426    }
427}

429/// A client that passed the handshake and waits for a decision: call
430/// [`accept`](Incoming::accept) to get its [`Socket`], or drop it to reject (the
431/// client's connect then fails with [`Error::Closed`]).
432pub struct Incoming {
433    conn: Option<Arc<sys::Conn>>,
434}
435
436impl Incoming {
437    /// Set the message handler of the future socket.
438    pub fn on_message(self, handler: impl FnMut(&[u8]) + Send + 'static) -> Self {
439        if let Some(c) = &self.conn {
440            c.set_message_handler(Some(Box::new(handler)));
441        }
442        self
443    }
444
445    /// Set the close handler of the future socket.
446    pub fn on_close(self, handler: impl FnOnce(CloseReason) + Send + 'static) -> Self {
447        if let Some(c) = &self.conn {
448            c.set_close_handler(Some(Box::new(handler)));
449        }
450        self
451    }
452
453    /// Accept the client.
454    pub fn accept(mut self) -> Socket {
455        let conn = self.conn.take().expect("accept consumes the Incoming");
456        conn.complete_accept();
457        Socket { conn }
458    }
459
460    /// Take the socket now and decide later. Handlers can be set and messages
461    /// queued on the socket right away, but the client's connect only completes
462    /// once [`PendingAccept::finish`] is called. Dropping the socket without
463    /// finishing rejects the client.
464    pub fn detach(mut self) -> (Socket, PendingAccept) {
465        let conn = self.conn.take().expect("detach consumes the Incoming");
466        (Socket { conn: conn.clone() }, PendingAccept { conn })
467    }
468}
469
470/// The decision still owed for a [`detach`](Incoming::detach)ed client.
471pub struct PendingAccept {
472    conn: Arc<sys::Conn>,
473}
474
475impl PendingAccept {
476    /// Complete the handshake: the client's connect returns.
477    pub fn finish(self) {
478        self.conn.complete_accept();
479    }
480}
481
482impl Drop for Incoming {
483    fn drop(&mut self) {
484        if let Some(c) = self.conn.take() {
485            c.close();
486        }
487    }
