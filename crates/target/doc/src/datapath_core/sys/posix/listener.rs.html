26const K_OFFER: i32 = 1;
27
28/// Called on an I/O thread for each client that passed the handshake. The
29/// callee decides: it either accepts the connection or drops it to reject.
30pub(crate) type AcceptHandler = Box<dyn FnMut(Arc<UdsConn>) + Send + 'static>;
31
32struct State {
33    fd: Option<OwnedFd>,
34    closed: bool,
35    unlink_on_close: bool,
36    connected: usize,
37    handler: Option<Arc<Mutex<AcceptHandler>>>,
38    /// Connections that passed the handshake before a handler was set.
39    waiting: VecDeque<Arc<UdsConn>>,
40    offer_posted: bool,
41    conns: Vec<Weak<UdsConn>>,
42}
43
44pub(crate) struct Listener {
45    me: Weak<Listener>,
46    ctx: Arc<ContextShared>,
47    loop_: Arc<Loop>,
48    addr: SocketAddress,
49    opts: Options,
50    /// Held while the accept handler runs; `release` takes it to wait it out.
51    callback_lock: ReentrantLock,
52    state: Mutex<State>,
53}
54
55/// True if the socket file at `addr` has no server behind it. Linux refuses
56/// connections only when nobody listens (a full backlog is EAGAIN); macOS also
57/// refuses them while a live server's backlog is full, so it asks a few times
58/// before declaring the file stale.
59fn stale_socket(addr: &SocketAddress) -> bool {
60    let attempts = if cfg!(target_os = "macos") { 20 } else { 1 };
61    for i in 0..attempts {
62        if i > 0 {
63            std::thread::sleep(std::time::Duration::from_millis(5));
64        }
65        let Ok(probe) = open_stream_socket() else {
66            return false;
67        };
68        if !matches!(
69            rustix::net::connect_unix(&probe, &addr.addr),
70            Err(Errno::CONNREFUSED)
71        ) {
72            return false;
73        }
74    }
75    true
76}

78impl Listener {
79    pub fn new(
80        ctx: Arc<ContextShared>,
81        loop_: Arc<Loop>,
82        addr: SocketAddress,
83        opts: Options,
84    ) -> Arc<Listener> {
85        Arc::new_cyclic(|me| Listener {
86            me: me.clone(),
87            ctx,
88            loop_,
89            addr,
90            opts,
91            callback_lock: ReentrantLock::new(),
92            state: Mutex::new(State {
93                fd: None,
94                closed: false,
95                unlink_on_close: false,
96                connected: 0,
97                handler: None,
98                waiting: VecDeque::new(),
99                offer_posted: false,
100                conns: Vec::new(),
101            }),
102        })
103    }
104
105    fn key(&self) -> u64 {
106        self as *const Listener as *const () as usize as u64
107    }
108
109    pub fn create(&self) -> Result<(), Error> {
110        let fd = open_stream_socket().map_err(|_| Error::CriticalFailure)?;
111        let do_bind = |fd: &OwnedFd| rustix::net::bind_unix(fd, &self.addr.addr);
112        if let Err(e) = do_bind(&fd) {
113            if e != Errno::ADDRINUSE {
114                return Err(if e == Errno::ACCESS {
115                    Error::AccessDenied
116                } else {
117                    Error::InvalidPath
118                });
119            }
120            let Some(path) = &self.addr.filesystem_path else {
121                return Err(Error::AddressInUse);
122            };
123            // A stale socket file is reused only if it is a socket nobody answers on.
124            match lstat(path.as_str()) {
125                Ok(st) if FileType::from_raw_mode(st.st_mode) == FileType::Socket => {}
126                _ => return Err(Error::AddressInUse),
127            }
128            if !stale_socket(&self.addr) {
129                return Err(Error::AddressInUse);
130            }
131            let _ = unlink(path.as_str());
132            if do_bind(&fd).is_err() {
133                return Err(Error::AddressInUse);
134            }
135        }
136        if let Some(path) = &self.addr.filesystem_path {
137            lock(&self.state).unlink_on_close = true;
138            let mode = match self.opts.permissions {
139                Permissions::User => 0o600,
140                Permissions::Group => 0o660,
141                Permissions::World => 0o666,
142            };
143            // Before listen(): nobody can connect until the mode is right.
144            if chmod(path.as_str(), Mode::from_bits_truncate(mode)).is_err() {
145                return Err(Error::AccessDenied);
146            }
147            // BSDs give a new file the group of its directory, not the creator's:
148            // Group access means the creator's group.
149            if self.opts.permissions == Permissions::Group
150                && chown(path.as_str(), None, Some(rustix::process::getegid())).is_err()
151            {
152                return Err(Error::AccessDenied);
153            }
154        }
155        rustix::net::listen(&fd, self.opts.backlog.max(16) as i32).map_err(|_| Error::Failure)?;
156        let me = self.me.upgrade().ok_or(Error::CriticalFailure)?;
157        let mut st = lock(&self.state);
158        let fd = st.fd.insert(fd);
159        if !self.loop_.add(&*fd, ev::IN, me) {
160            return Err(Error::CriticalFailure);
161        }
162        Ok(())
163    }
164
165    fn peer_allowed(&self, fd: &OwnedFd) -> bool {
166        if !self.addr.is_abstract() || self.opts.permissions == Permissions::World {
167            return true; // filesystem sockets are guarded by their mode bits
168        }
169        let Some((uid, gid)) = peer_credentials(fd) else {
170            return false;
171        };
172        if uid == 0 || uid == rustix::process::geteuid().as_raw() {
173            return true;
174        }
175        self.opts.permissions == Permissions::Group && gid == rustix::process::getegid().as_raw()
176    }
177
178    /// A client passed the handshake: hand it to the accept handler.
179    pub fn offer(&self, conn: Arc<UdsConn>) {
180        let handler = {
181            let mut st = lock(&self.state);
182            if st.closed {
183                drop(st);
184                conn.close();
185                return;
186            }
187            match &st.handler {
188                None => {
189                    st.waiting.push_back(conn);
190                    return;
191                }
192                Some(h) => h.clone(),
193            }
194        };
195        let _cb = self.callback_lock.lock();
196        if lock(&self.state).closed {
197            conn.close();
198            return;
199        }
200        // If the handler panics, `conn` is dropped on the way out, which rejects it.
201        let _ = catch_unwind(AssertUnwindSafe(|| (*lock(&handler))(conn)));
202    }
203
204    pub fn connection_finished(&self) {
205        let mut st = lock(&self.state);
206        st.connected = st.connected.saturating_sub(1);
207    }
208
209    pub fn set_accept_handler(&self, handler: Option<AcceptHandler>) {
210        let old = {
211            let mut st = lock(&self.state);
212            let old = std::mem::replace(&mut st.handler, handler.map(|f| Arc::new(Mutex::new(f))));
213            if st.handler.is_some() && !st.waiting.is_empty() && !st.offer_posted {
214                if let Some(me) = self.me.upgrade() {
215                    st.offer_posted = self.loop_.post(Some(me), K_OFFER);
216                }
217            }
218            old
219        };
220        // The old handler may own sockets: drop it outside the lock.
221        drop(old);
222    }
223
224    pub fn close(&self) -> bool {
225        let (conns, waiting, handler) = {
226            let mut st = lock(&self.state);
227            if st.closed {
228                return false;
229            }
230            st.closed = true;
231            if let Some(fd) = st.fd.take() {
232                self.loop_.remove_fd(&fd);
233            }
234            if st.unlink_on_close {
235                if let Some(path) = &self.addr.filesystem_path {
236                    let _ = unlink(path.as_str());
237                }
238            }
239            (
240                std::mem::take(&mut st.conns),
241                std::mem::take(&mut st.waiting),
242                st.handler.take(),
243            )
244        };
245        // The handler may own sockets: drop it outside the lock.
246        drop(handler);
247        for c in waiting {
248            c.close();
249        }
250        for c in conns.iter().filter_map(Weak::upgrade) {
251            c.close();
252        }
253        // Drop the loop's reference on the loop thread.
254        if let Some(me) = self.me.upgrade() {
255            self.loop_.post(Some(me), 0);
256        }
257        true
258    }
259
260    /// Close, and wait for a running accept handler.
261    pub fn release(&self) {
262        self.close();
263        drop(self.callback_lock.lock());
264    }
265}
266
267impl Target for Listener {
268    fn on_event(&self, _events: u32) {
269        loop {
270            let accepted = {
271                let st = lock(&self.state);
272                match &st.fd {
273                    Some(fd) if !st.closed => accept_stream_socket(fd),
274                    _ => return,
275                }
276            };
277            let fd = match accepted {
278                Ok(fd) => fd,
279                Err(Errno::INTR) | Err(Errno::CONNABORTED) => continue,
280                // EAGAIN, or out of descriptors: wait for the next edge.
281                Err(_) => return,
282            };
283            let conn = {
284                let mut st = lock(&self.state);
285                let full = self.opts.max_clients != 0 && st.connected >= self.opts.max_clients;
286                if st.closed || full || !self.peer_allowed(&fd) {
287                    continue;
288                }
289                st.connected += 1;
290                tune_socket_buffers(&fd, &self.opts);
291                let conn = UdsConn::new(
292                    self.ctx.clone(),
293                    self.ctx.pick_loop(),
294                    fd,
295                    &self.opts,
296                    self.opts.max_message_size,
297                );
298                st.conns.retain(|w| w.strong_count() > 0);
299                st.conns.push(Arc::downgrade(&conn));
300                conn
301            };
302            self.ctx.add_connection(Arc::downgrade(&conn));
303            if !conn.start(true, self.me.clone(), self.opts.max_message_size) {
304                conn.close();
305            }
306        }
307    }
308
309    fn on_post(&self, kind: i32) {
310        if kind == 0 {
311            self.loop_.unregister(self.key());
312            return;
313        }
314        if kind != K_OFFER {
315            return;
316        }
317        loop {
318            let conn = {
319                let mut st = lock(&self.state);
320                if st.waiting.is_empty() || st.handler.is_none() {
321                    st.offer_posted = false;
322                    return;
323                }
324                st.waiting.pop_front()
325            };
326            if let Some(c) = conn {
327                self.offer(c);
328            }
329        }
330    }
