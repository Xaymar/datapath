6//! epoll, woken by an eventfd (Linux, Android).
7
8use crate::sys::posix::ev;
9use rustix::event::epoll::{self, CreateFlags, EventData, EventFlags, EventVec};
10use rustix::event::{eventfd, EventfdFlags};
11use std::os::fd::{AsFd, OwnedFd};
12
13/// Key of the wake-up eventfd; target keys are addresses and never 0.
14pub(crate) const WAKE_KEY: u64 = 0;
15
16pub(crate) struct Poller {
17    epoll: OwnedFd,
18    wake_fd: OwnedFd,
19}
20
21pub(crate) type Events = EventVec;
22
23fn to_flags(events: u32) -> EventFlags {
24    let mut r = EventFlags::ET;
25    if events & ev::IN != 0 {
26        r |= EventFlags::IN;
27    }
28    if events & ev::OUT != 0 {
29        r |= EventFlags::OUT;
30    }
31    if events & ev::RDHUP != 0 {
32        r |= EventFlags::RDHUP;
33    }
34    r
35}
36
37fn from_flags(f: EventFlags) -> u32 {
38    let mut r = 0;
39    if f.contains(EventFlags::IN) {
40        r |= ev::IN;
41    }
42    if f.contains(EventFlags::OUT) {
43        r |= ev::OUT;
44    }
45    if f.contains(EventFlags::RDHUP) {
46        r |= ev::RDHUP;
47    }
48    if f.contains(EventFlags::HUP) {
49        r |= ev::HUP;
50    }
51    if f.contains(EventFlags::ERR) {
52        r |= ev::ERR;
53    }
54    r
55}
56
57impl Poller {
58    pub fn new() -> std::io::Result<Poller> {
59        let epoll = epoll::create(CreateFlags::CLOEXEC)?;
60        let wake_fd = eventfd(0, EventfdFlags::NONBLOCK | EventfdFlags::CLOEXEC)?;
61        epoll::add(
62            &epoll,
63            &wake_fd,
64            EventData::new_u64(WAKE_KEY),
65            EventFlags::IN,
66        )?;
67        Ok(Poller { epoll, wake_fd })
68    }
69
70    pub fn events() -> Events {
71        EventVec::with_capacity(64)
72    }
73
74    /// Register `fd` (edge-triggered) for `events`.
75    pub fn add(&self, fd: impl AsFd, events: u32, key: u64) -> bool {
76        epoll::add(&self.epoll, fd, EventData::new_u64(key), to_flags(events)).is_ok()
77    }
78
79    /// Change the registration; this re-checks readiness, so enabling `OUT` on
80    /// a socket that is already writable reports it.
81    pub fn modify(&self, fd: impl AsFd, _old_events: u32, events: u32, key: u64) -> bool {
82        epoll::modify(&self.epoll, fd, EventData::new_u64(key), to_flags(events)).is_ok()
83    }
84
85    pub fn remove(&self, fd: impl AsFd) {
86        let _ = epoll::delete(&self.epoll, fd);
87    }
88
89    pub fn wake(&self) {
90        let _ = rustix::io::write(&self.wake_fd, &1u64.to_ne_bytes());
91    }
92
93    /// Consume a wake-up.
94    pub fn drain_wake(&self) {
95        let mut buf = [0u8; 8];
96        let _ = rustix::io::read(&self.wake_fd, &mut buf);
97    }
98
99    /// Wait up to `timeout_ms` (-1 = forever, 0 = poll) and fill `out` with
100    /// (key, `ev` bits). Returns how many events there are.
101    pub fn wait(&self, timeout_ms: i32, events: &mut Events, out: &mut Vec<(u64, u32)>) -> usize {
102        out.clear();
103        if epoll::wait(&self.epoll, events, timeout_ms).is_err() {
104            return 0;
105        }
106        for e in events.iter() {
107            out.push((e.data.u64(), from_flags(e.flags)));
108        }
109        out.len()
110    }
111}
