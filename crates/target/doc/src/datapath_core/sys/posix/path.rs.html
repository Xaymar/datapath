6//! Server paths to socket addresses.
7
8use rustix::net::SocketAddrUnix;
9
10/// A resolved socket address.
11pub(crate) struct SocketAddress {
12    pub addr: SocketAddrUnix,
13    /// The socket file, for filesystem sockets.
14    pub filesystem_path: Option<String>,
15}
16
17impl SocketAddress {
18    pub fn is_abstract(&self) -> bool {
19        self.filesystem_path.is_none()
20    }
21}
22
23/// An absolute path is used as it is. A bare name lives in the abstract
24/// namespace on Linux and Android, and in `/tmp/datapath.<name>` elsewhere
25/// (which every user shares, like the abstract namespace; the per-user
26/// `$TMPDIR` would hide a server from other users' clients).
27pub(crate) fn make_socket_address(path: &str) -> Option<SocketAddress> {
28    if path.is_empty() || path.contains('\0') {
29        return None;
30    }
31    if path.starts_with('/') {
32        let addr = SocketAddrUnix::new(path).ok()?;
33        return Some(SocketAddress {
34            addr,
35            filesystem_path: Some(path.to_owned()),
36        });
37    }
38    #[cfg(any(target_os = "linux", target_os = "android"))]
39    {
40        let name = format!("datapath.{path}");
41        let addr = SocketAddrUnix::new_abstract_name(name.as_bytes()).ok()?;
42        Some(SocketAddress {
43            addr,
44            filesystem_path: None,
45        })
46    }
47    #[cfg(not(any(target_os = "linux", target_os = "android")))]
48    {
49        make_socket_address(&format!("/tmp/datapath.{path}"))
50    }
51}

53#[cfg(test)]
54mod tests {
55    use super::*;
56
57    #[test]
58    fn absolute_paths_are_filesystem_sockets() {
59        let a = make_socket_address("/tmp/x.sock").unwrap();
60        assert_eq!(a.filesystem_path.as_deref(), Some("/tmp/x.sock"));
61        assert!(!a.is_abstract());
62    }
63
64    #[test]
65    fn rejects_empty_and_too_long_paths() {
66        assert!(make_socket_address("").is_none());
67        assert!(make_socket_address(&format!("/{}", "a".repeat(200))).is_none());
68        assert!(make_socket_address("bad\0name").is_none());
69    }
70
71    #[cfg(any(target_os = "linux", target_os = "android"))]
72    #[test]
73    fn bare_names_are_abstract_on_linux() {
74        assert!(make_socket_address("my-service").unwrap().is_abstract());
75    }
