6//! One I/O thread: a poller, the targets registered with it, and a queue of
7//! notifications posted from other threads.
8
9use super::context::ContextShared;
10use super::platform::apply_thread_priority;
11use super::poller::{Events, Poller, WAKE_KEY};
12use super::{key_of, Target};
13use crate::conn::lock;
14use crate::spin::spin_then_block;
15use std::cell::{Cell, RefCell};
16use std::collections::HashMap;
17use std::os::fd::AsFd;
18use std::sync::{Arc, Mutex};
19use std::thread::JoinHandle;
20use std::time::Duration;
21
22thread_local! {
23    /// The context whose I/O thread is running on this thread, if any.
24    static CURRENT_CONTEXT: Cell<usize> = const { Cell::new(0) };
25}
26
27/// True on any I/O thread of the context with this id.
28pub(crate) fn on_io_thread_of(ctx_id: usize) -> bool {
29    CURRENT_CONTEXT.with(|c| c.get() == ctx_id)
30}
31
32struct Posts {
33    /// `None` target = quit.
34    items: Vec<(Option<Arc<dyn Target>>, i32)>,
35    signaled: bool,
36    /// The thread has been joined: posts are refused.
37    dead: bool,
38}
39
40pub(crate) struct Loop {
41    poller: Poller,
42    posts: Mutex<Posts>,
43    registered: Mutex<HashMap<u64, Arc<dyn Target>>>,
44    thread: Mutex<Option<JoinHandle<()>>>,
45}
46
47impl Loop {
48    pub fn new() -> std::io::Result<Arc<Loop>> {
49        Ok(Arc::new(Loop {
50            poller: Poller::new()?,
51            posts: Mutex::new(Posts {
52                items: Vec::new(),
53                signaled: false,
54                dead: false,
55            }),
56            registered: Mutex::new(HashMap::new()),
57            thread: Mutex::new(None),
58        }))
59    }
60
61    /// Start the thread. It keeps `ctx` alive, so a context is never destroyed
62    /// on one of its own I/O threads.
63    pub fn start(self: &Arc<Self>, ctx: Arc<ContextShared>) -> std::io::Result<()> {
64        let me = self.clone();
65        let handle = std::thread::Builder::new()
66            .name("datapath-io".into())
67            .spawn(move || {
68                CURRENT_CONTEXT.with(|c| c.set(ctx.id()));
69                apply_thread_priority(ctx.options().priority.level());
70                me.run(&ctx);
71            })?;
72        *lock(&self.thread) = Some(handle);
73        Ok(())
74    }
75
76    /// Register `fd` (edge-triggered) for `events`. The loop keeps `target`
77    /// alive until [`unregister`](Self::unregister).
78    pub fn add(&self, fd: impl AsFd, events: u32, target: Arc<dyn Target>) -> bool {
79        let key = key_of(&target);
80        lock(&self.registered).insert(key, target);
81        if self.poller.add(fd, events, key) {
82            true
83        } else {
84            lock(&self.registered).remove(&key);
85            false
86        }
87    }
88
89    pub fn modify(&self, fd: impl AsFd, old_events: u32, events: u32, key: u64) -> bool {
90        self.poller.modify(fd, old_events, events, key)
91    }
92
93    pub fn remove_fd(&self, fd: impl AsFd) {
94        self.poller.remove(fd);
95    }
96
97    /// Drop the loop's reference to the target with this key.
98    pub fn unregister(&self, key: u64) {
99        let dying = lock(&self.registered).remove(&key);
100        // `dying` may be the last reference; it is released outside the lock.
101        drop(dying);
102    }
103
104    /// Run `target.on_post(kind)` on this loop's thread. False if the thread is gone.
105    pub fn post(&self, target: Option<Arc<dyn Target>>, kind: i32) -> bool {
106        let signal;
107        {
108            let mut p = lock(&self.posts);
109            if p.dead {
110                return false;
111            }
112            p.items.push((target, kind));
113            signal = !p.signaled;
114            p.signaled = true;
115        }
116        if signal {
117            self.poller.wake();
118        }
119        true
120    }
121
122    fn drain_posts(&self, quit: &Cell<bool>) -> bool {
123        let items = {
124            let mut p = lock(&self.posts);
125            p.signaled = false;
126            std::mem::take(&mut p.items)
127        };
128        let any = !items.is_empty();
129        for (target, kind) in items {
130            match target {
131                None => quit.set(true),
132                Some(t) => t.on_post(kind),
133            }
134        }
135        any
136    }
137
138    /// Wait up to `timeout` ms and dispatch. False if nothing happened.
139    fn pump(
140        &self,
141        timeout: i32,
142        events: &RefCell<(Events, Vec<(u64, u32)>)>,
143        quit: &Cell<bool>,
144    ) -> bool {
145        let mut state = events.borrow_mut();
146        let (evs, out) = &mut *state;
147        if self.poller.wait(timeout, evs, out) == 0 {
148            return false;
149        }
150        let mut posts = false;
151        for &(key, bits) in out.iter() {
152            if key == WAKE_KEY {
153                posts = true;
154                continue;
155            }
156            let target = lock(&self.registered).get(&key).cloned();
157            if let Some(t) = target {
158                t.on_event(bits);
159            }
160        }
161        drop(state);
162        if posts {
163            self.poller.drain_wake();
164            self.drain_posts(quit);
165        }
166        true
167    }
168
169    fn run(&self, ctx: &ContextShared) {
170        let events = RefCell::new((Poller::events(), Vec::new()));
171        let quit = Cell::new(false);
172        let spin: Duration = ctx.options().spin;
173        let yield_cpu = ctx.options().yield_cpu;
174        while !quit.get() {
175            spin_then_block(
176                spin,
177                || !quit.get() && self.pump(0, &events, &quit),
178                || {
179                    if !quit.get() {
180                        self.pump(-1, &events, &quit);
181                    }
182                },
183                || {
184                    if yield_cpu {
185                        std::thread::yield_now();
186                    } else {
187                        crate::spin::cpu_relax();
188                    }
189                },
190            );
191        }
192    }
193
194    /// Ask the thread to quit after the notifications queued so far.
195    pub fn request_quit(&self) {
196        self.post(None, 0);
197    }
198
199    /// Join the thread, then run notifications that raced the quit request
200    /// (for example `on_close`), and drop every registration to break
201    /// context <-> target reference cycles.
202    pub fn finish(&self) {
203        let handle = lock(&self.thread).take();
204        if let Some(h) = handle {
205            let _ = h.join();
206        }
207        let items = {
208            let mut p = lock(&self.posts);
209            p.dead = true;
210            std::mem::take(&mut p.items)
211        };
212        for (target, kind) in items {
213            if let Some(t) = target {
214                t.on_post(kind);
215            }
216        }
217        let regs = std::mem::take(&mut *lock(&self.registered));
218        drop(regs);
219    }
220}
