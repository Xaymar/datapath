6//! The wire format: an 8-byte little-endian length followed by the payload, after
7//! a 16-byte handshake in each direction. Identical to the C++ core, so a Rust
8//! and a C++ peer can talk to each other.
9
10use crate::error::Error;
11
12/// Size of the length prefix of every message.
13pub const FRAME_HEADER_SIZE: usize = 8;
14/// Protocol version carried in the handshake.
15pub const PROTOCOL_VERSION: u32 = 1;
16/// Size of the handshake.
17pub const HELLO_SIZE: usize = 16;
18
19/// The frame header for a payload of `length` bytes.
20pub fn encode_frame_header(length: u64) -> [u8; FRAME_HEADER_SIZE] {
21    length.to_le_bytes()
22}
23
24/// The connection handshake: `"DPTH"`, a `u32` version and the `u64` largest
25/// message this side accepts, little-endian.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub struct Hello {
28    /// Protocol version of the sender.
29    pub version: u32,
30    /// Largest message the sender accepts.
31    pub max_message_size: u64,
32}
33
34impl Hello {
35    /// A handshake for the current protocol version.
36    pub fn new(max_message_size: u64) -> Hello {
37        Hello {
38            version: PROTOCOL_VERSION,
39            max_message_size,
40        }
41    }
42
43    /// The bytes to send.
44    pub fn encode(&self) -> [u8; HELLO_SIZE] {
45        let mut out = [0u8; HELLO_SIZE];
46        out[..4].copy_from_slice(b"DPTH");
47        out[4..8].copy_from_slice(&self.version.to_le_bytes());
48        out[8..].copy_from_slice(&self.max_message_size.to_le_bytes());
49        out
50    }
51
52    /// Parse a received handshake. A wrong magic or a version other than
53    /// [`PROTOCOL_VERSION`] is [`Error::VersionMismatch`].
54    pub fn decode(bytes: &[u8; HELLO_SIZE]) -> Result<Hello, Error> {
55        if &bytes[..4] != b"DPTH" {
56            return Err(Error::VersionMismatch);
57        }
58        let version = u32::from_le_bytes(bytes[4..8].try_into().expect("4 bytes"));
59        let max_message_size = u64::from_le_bytes(bytes[8..].try_into().expect("8 bytes"));
60        if version != PROTOCOL_VERSION {
61            return Err(Error::VersionMismatch);
62        }
63        Ok(Hello {
64            version,
65            max_message_size,
66        })
67    }
68}

70/// Smallest read the parser asks for before it has seen larger traffic.
71const MIN_READ: usize = 64 * 1024;
72
73/// Incremental, zero-copy frame parser for a byte stream.
74///
75/// ```
76/// use datapath_core::proto::{encode_frame_header, FrameParser};
77///
78/// let mut parser = FrameParser::new(1 << 20, 4096);
79/// let mut wire = encode_frame_header(5).to_vec();
80/// wire.extend_from_slice(b"hello");
81///
82/// let room = parser.prepare();           // read up to room.len() bytes into room
83/// room[..wire.len()].copy_from_slice(&wire);
84/// let mut seen = Vec::new();
85/// parser.commit(wire.len(), |msg| { seen.push(msg.to_vec()); true }).unwrap();
86/// assert_eq!(seen, [b"hello".to_vec()]);
87/// ```
88///
89/// Small frames are delivered straight out of the receive buffer. A frame larger
90/// than the buffer gets an exact-size buffer of its own, which the following
91/// reads fill directly and which is released when a small frame arrives again.
92pub struct FrameParser {
93    buf: Vec<u8>,
94    rpos: usize,
95    wpos: usize,
96    big: Vec<u8>,
97    big_filled: usize,
98    in_big: bool,
99    limit: u64,
100    // Adaptive read size: small traffic keeps reads small (large read requests are
101    // expensive for Windows pipes even when little data arrives); reads that fill
102    // the request completely double it, up to the buffer size.
103    read_hint: usize,
104    last_request: usize,
105}
106
107impl FrameParser {
108    /// A parser that rejects frames above `max_message_size` and receives into a
109    /// buffer of `buffer_size` bytes (at least 64).
110    pub fn new(max_message_size: u64, buffer_size: usize) -> FrameParser {
111        let size = buffer_size.max(64);
112        FrameParser {
113            buf: vec![0; size],
114            rpos: 0,
115            wpos: 0,
116            big: Vec::new(),
117            big_filled: 0,
118            in_big: false,
119            limit: max_message_size,
120            read_hint: size.min(MIN_READ),
121            last_request: 0,
122        }
123    }
124
125    /// Change the largest accepted frame (after the handshake negotiated it).
126    pub fn set_limit(&mut self, max_message_size: u64) {
127        self.limit = max_message_size;
128    }
129
130    fn compact(&mut self) {
131        if self.rpos == 0 {
132            return;
133        }
134        self.buf.copy_within(self.rpos..self.wpos, 0);
135        self.wpos -= self.rpos;
136        self.rpos = 0;
137    }
138
139    /// Where the next read should go. Pass the number of bytes actually read to
140    /// [`commit`](Self::commit).
141    pub fn prepare(&mut self) -> &mut [u8] {
142        if self.in_big {
143            let filled = self.big_filled;
144            return &mut self.big[filled..];
145        }
146        if self.rpos == self.wpos {
147            self.rpos = 0;
148            self.wpos = 0;
149        } else if self.wpos == self.buf.len() || self.buf.len() - self.wpos < FRAME_HEADER_SIZE {
150            self.compact();
151        }
152        let room = self.buf.len() - self.wpos;
153        self.last_request = room.min(self.read_hint);
154        let start = self.wpos;
155        &mut self.buf[start..start + self.last_request]
156    }
157
158    /// Account for `n` bytes read into the slice returned by
159    /// [`prepare`](Self::prepare) and call `deliver` for every complete frame.
160    ///
161    /// `deliver` returns `false` to stop parsing (the connection was closed from
162    /// a handler), which yields [`Error::Closed`]. A frame above the limit is
163    /// [`Error::TooLarge`]; failing to allocate a large frame is
164    /// [`Error::OutOfMemory`].
165    ///
166    /// # Panics
167    /// If `n` is larger than the slice `prepare` returned.
168    pub fn commit<F>(&mut self, n: usize, mut deliver: F) -> Result<(), Error>
169    where
170        F: FnMut(&[u8]) -> bool,
171    {
172        if self.in_big {
173            assert!(
174                n <= self.big.len() - self.big_filled,
175                "commit larger than prepare"
176            );
177            self.big_filled += n;
178            if self.big_filled < self.big.len() {
179                return Ok(());
180            }
181            self.in_big = false;
182            self.big_filled = 0;
183            return if deliver(&self.big) {
184                Ok(())
185            } else {
186                Err(Error::Closed)
187            };
188        }
189
190        assert!(n <= self.last_request, "commit larger than prepare");
191        self.wpos += n;
192        if n == self.last_request && self.read_hint < self.buf.len() {
193            self.read_hint = (self.read_hint * 2).min(self.buf.len());
194        } else if n < self.read_hint / 8 && self.read_hint > MIN_READ {
195            self.read_hint /= 2;
196        }
197
198        loop {
199            let avail = self.wpos - self.rpos;
200            if avail < FRAME_HEADER_SIZE {
201                break;
202            }
203            let header: [u8; FRAME_HEADER_SIZE] = self.buf
204                [self.rpos..self.rpos + FRAME_HEADER_SIZE]
205                .try_into()
206                .expect("8 bytes");
207            let len = u64::from_le_bytes(header);
208            if len > self.limit {
209                return Err(Error::TooLarge);
210            }
211            // A length that does not fit usize can never fit the buffer either, and
212            // is only reachable on 32-bit targets with a limit above 4 GiB.
213            let len = usize::try_from(len).map_err(|_| Error::TooLarge)?;
214            let have = avail - FRAME_HEADER_SIZE;
215            if have >= len {
216                if !self.big.is_empty() {
217                    // Back to small frames: give the large buffer back.
218                    self.big = Vec::new();
219                }
220                let start = self.rpos + FRAME_HEADER_SIZE;
221                self.rpos = start + len;
222                if !deliver(&self.buf[start..start + len]) {
223                    return Err(Error::Closed);
224                }
225                continue;
226            }
227            if FRAME_HEADER_SIZE + len <= self.buf.len() {
228                // Fits the buffer; make room for the rest and wait for more data.
229                if self.rpos + FRAME_HEADER_SIZE + len > self.buf.len() {
230                    self.compact();
231                }
232                break;
233            }
234            // Larger than the receive buffer: move to a dedicated buffer, reusing
235            // the one of the previous large frame when it is big enough.
236            if self.big.capacity() < len {
237                self.big = Vec::new();
238                self.big
239                    .try_reserve_exact(len)
240                    .map_err(|_| Error::OutOfMemory)?;
241            }
242            self.big.clear();
243            self.big.resize(len, 0);
244            let start = self.rpos + FRAME_HEADER_SIZE;
245            self.big[..have].copy_from_slice(&self.buf[start..start + have]);
246            self.big_filled = have;
247            self.rpos = 0;
248            self.wpos = 0;
249            self.in_big = true;
250            break;
251        }
252        Ok(())
253    }
254}

260    fn frame(payload: &[u8]) -> Vec<u8> {
261        let mut v = encode_frame_header(payload.len() as u64).to_vec();
262        v.extend_from_slice(payload);
263        v
264    }
265
266    /// Feed `wire` to the parser in chunks of `chunk` bytes and collect the frames.
267    fn feed(parser: &mut FrameParser, wire: &[u8], chunk: usize) -> Result<Vec<Vec<u8>>, Error> {
268        let mut out = Vec::new();
269        let mut pos = 0;
270        while pos < wire.len() {
271            let room = parser.prepare();
272            let n = room.len().min(chunk).min(wire.len() - pos);
273            room[..n].copy_from_slice(&wire[pos..pos + n]);
274            pos += n;
275            parser.commit(n, |m| {
276                out.push(m.to_vec());
277                true
278            })?;
279        }
280        Ok(out)
281    }

316    #[test]
317    fn frames_split_at_every_byte_offset() {
318        let payloads: [&[u8]; 4] = [b"", b"a", b"hello world", &[7u8; 200]];
319        let mut wire = Vec::new();
320        for p in payloads {
321            wire.extend_from_slice(&frame(p));
322        }
323        // Every way to cut the stream in two reads, then byte-by-byte.
324        for cut in 0..=wire.len() {
325            let mut parser = FrameParser::new(1 << 20, 64);
326            let mut got = feed(&mut parser, &wire[..cut], usize::MAX).unwrap();
327            got.extend(feed(&mut parser, &wire[cut..], usize::MAX).unwrap());
328            assert_eq!(
329                got,
330                payloads.iter().map(|p| p.to_vec()).collect::<Vec<_>>(),
331                "cut at {cut}"
332            );
333        }
334        let mut parser = FrameParser::new(1 << 20, 64);
335        let got = feed(&mut parser, &wire, 1).unwrap();
336        assert_eq!(got.len(), 4);
337    }

426    #[test]
427    fn read_size_adapts_to_traffic() {
428        let mut parser = FrameParser::new(1 << 20, 256 * 1024);
429        assert_eq!(parser.prepare().len(), MIN_READ);
430        // A read that fills the request doubles the next one.
431        let n = parser.prepare().len();

433        assert_eq!(parser.prepare().len(), 2 * MIN_READ);
