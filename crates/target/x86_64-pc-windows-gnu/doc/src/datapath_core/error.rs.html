6//! Error codes. The numeric values are the ones of `datapath_error` in the C API
7//! (`datapath.h`) and of `datapath::error` in the C++ API, so they can be passed
8//! across the C ABI unchanged.
9
10use std::fmt;
11
12/// Why an operation failed. `Success` (0) is not an error and has no variant.
13#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
14#[repr(i32)]
15pub enum Error {
16    /// Generic failure; details were not mappable to a specific code.
17    Failure = 1,
18    /// A failure the library cannot recover from.
19    CriticalFailure = 2,
20    /// The connection or server is closed.
21    Closed = 3,
22    /// The operation did not finish in time.
23    TimedOut = 4,
24    /// The path is malformed or too long for the platform.
25    InvalidPath = 5,
26    /// The platform does not support this.
27    NotSupported = 6,
28    /// The message exceeds the negotiated maximum message size.
29    TooLarge = 7,
30    /// The send queue is full; retry later.
31    WouldBlock = 8,
32    /// The peer speaks a different protocol version.
33    VersionMismatch = 9,
34    /// Nothing is listening at the path.
35    NotFound = 10,
36    /// Another server already hosts this path.
37    AddressInUse = 11,
38    /// The operating system denied access.
39    AccessDenied = 12,
40    /// A required value was missing or out of range.
41    InvalidArgument = 13,
42    /// Memory allocation failed.
43    OutOfMemory = 14,
44}
45
46impl Error {
47    /// The C API code of this error.
48    pub fn code(self) -> i32 {
49        self as i32
50    }
51
52    /// The error for a C API code; `None` for 0 (success), -1 (unknown) and codes
53    /// this version does not know.
54    pub fn from_code(code: i32) -> Option<Error> {
55        Some(match code {
56            1 => Error::Failure,
57            2 => Error::CriticalFailure,
58            3 => Error::Closed,
59            4 => Error::TimedOut,
60            5 => Error::InvalidPath,
61            6 => Error::NotSupported,
62            7 => Error::TooLarge,
63            8 => Error::WouldBlock,
64            9 => Error::VersionMismatch,
65            10 => Error::NotFound,
66            11 => Error::AddressInUse,
67            12 => Error::AccessDenied,
68            13 => Error::InvalidArgument,
69            14 => Error::OutOfMemory,
70            _ => return None,
71        })
72    }
73}
74
75impl fmt::Display for Error {
76    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
77        f.write_str(match self {
78            Error::Failure => "operation failed",
79            Error::CriticalFailure => "critical failure",
80            Error::Closed => "connection closed",
81            Error::TimedOut => "timed out",
82            Error::InvalidPath => "invalid path",
83            Error::NotSupported => "not supported",
84            Error::TooLarge => "message too large",
85            Error::WouldBlock => "send queue full",
86            Error::VersionMismatch => "protocol version mismatch",
87            Error::NotFound => "nothing is listening at the path",
88            Error::AddressInUse => "address already in use",
89            Error::AccessDenied => "access denied",
90            Error::InvalidArgument => "invalid argument",
91            Error::OutOfMemory => "out of memory",
92        })
93    }
94}
95
96impl std::error::Error for Error {}

98#[cfg(test)]
99mod tests {
100    use super::*;
101
102    #[test]
103    fn codes_round_trip() {
104        for code in 1..=14 {
105            let e = Error::from_code(code).expect("known code");
106            assert_eq!(e.code(), code);
107        }
108        assert_eq!(Error::from_code(0), None);
109        assert_eq!(Error::from_code(15), None);
110        assert_eq!(Error::from_code(-1), None);
111    }
112
113    #[test]
114    fn codes_match_the_c_api() {
115        assert_eq!(Error::Closed.code(), 3);
116        assert_eq!(Error::WouldBlock.code(), 8);
117        assert_eq!(Error::NotFound.code(), 10);
118    }
119}
